! SW1 - phase 1: VLANs, access ports, trunk, management address
! In the SW1 CLI type:  enable   then   configure terminal   then paste everything below.
hostname SW1
no ip domain-lookup
!
vlan 10
 name STAFF
vlan 20
 name SERVERS
vlan 30
 name GUEST
vlan 99
 name MGMT
vlan 999
 name UNUSED
exit
!
interface fastEthernet0/1
 description SRV1
 switchport mode access
 switchport access vlan 20
interface fastEthernet0/2
 description PC1-STAFF
 switchport mode access
 switchport access vlan 10
interface fastEthernet0/3
 description PC2-STAFF
 switchport mode access
 switchport access vlan 10
interface fastEthernet0/4
 description PC3-GUEST
 switchport mode access
 switchport access vlan 30
!
interface gigabitEthernet0/1
 description TRUNK-TO-R1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,99
!
! Unused ports: park in a dead VLAN and shut down
interface range fastEthernet0/5 - 24
 switchport mode access
 switchport access vlan 999
 shutdown
interface gigabitEthernet0/2
 switchport mode access
 switchport access vlan 999
 shutdown
!
! Management address
interface vlan 99
 ip address 192.168.99.2 255.255.255.0
 no shutdown
exit
ip default-gateway 192.168.99.1
end
write memory
